% on the Gabriel Sison / sb2nov resume template (MIT). Selected with
% `--template=overleaf`; the Charter-based cv-template.tex remains the default.
%
% One thing changed in the port. The emitter writes \roleheading /
% \projheading / \bul / \bullist, not the sb2nov \resume* family. Both are
% defined below: \resume* because the design is built on them and
% generate-latex.mjs validates for them, and \roleheading et al. as thin
% wrappers so the emitter's output lands in the right shapes. Keep both.
%
% The original's faIcon contact row was dropped. build-cv-latex.mjs's
% CONTACT_LINE already emits email, LinkedIn, GitHub and portfolio as $|$-
% separated hrefs, plus any free-text phone/city line — so the icon row
% repeated all three (and double-prefixed the mailto:). One complete line beats
% a prettier duplicated one.
%
% The visual design — margins, \scshape rules, type scale — is unchanged from
% the original. Do not shrink type or margins to fit content; see
% modes/resume-authoring.md.
%------------------------

%%%%%%%%%%%%%%%%%%%%%%%%%%%%  Commands  %%%%%%%%%%%%%%%%%%%%%%%%%%%%
\newcommand{\resumeItem}[1]{
  \item\small{
    {#1 \vspace{-3pt}}
  }
}
\newcommand{\resumeSubheading}[4]{
  \vspace{-3pt}\item
    \begin{tabular*}{1.0\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{#1} & \textbf{\small #2} \\
      \textit{\small#3} & \textit{\small #4} \\
    \end{tabular*}\vspace{-7pt}
}
\newcommand{\resumeSubheadingContinue}[2]{
  \vspace{-3pt}
    \begin{tabular*}{1.0\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textit{\small#1} & \textit{\small #2} \\
    \end{tabular*}\vspace{-7pt}
}
\newcommand{\resumeProjectHeading}[2]{
  \vspace{-3pt}\item
    \begin{tabular*}{1.0\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{#1} & \textbf{\small #2} \\
    \end{tabular*}\vspace{-7pt}
}
\newcommand{\resumeSubItem}[1]{\resumeItem{#1}\vspace{0pt}}
\renewcommand\labelitemi{$\vcenter{\hbox{\tiny$\bullet$}}$}
\renewcommand\labelitemii{$\vcenter{\hbox{\tiny$\bullet$}}$}
\newcommand{\resumeSubHeadingListStart}{\begin{itemize}[leftmargin=0.0in, label={}]}
\newcommand{\resumeSubHeadingListEnd}{\end{itemize}}
\newcommand{\resumeItemListStart}{\begin{itemize}}
\newcommand{\resumeItemListEnd}{\end{itemize}\vspace{0pt}}

%%%%%%%%%%%%%%%%%%  build-cv-latex.mjs emitter contract  %%%%%%%%%%%%%%%%%%
% The emitter writes these four; each maps onto an sb2nov macro above so the
% design stays identical. Argument order below is the emitter's, which is
% {role}{dates}{company}{location} — note it differs from \resumeSubheading's
% {org}{dates}{title}{location}, so the swap happens here rather than
% anywhere it could be missed.
\newcommand{\roleheading}[4]{\resumeSubheading{#3}{#2}{#1}{#4}}
\newcommand{\projheading}[2]{\resumeProjectHeading{#1}{#2}}
\newcommand{\bul}[1]{\resumeItem{#1}}
\newcommand{\bullist}{\resumeItemListStart}
\newcommand{\stopbulls}{\resumeItemListEnd}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%  Heading  %%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{document}
    \begin{center}
        {\Huge\scshape {{NAME}}}
        \\ \vspace{3pt}

    \end{center}
    \vspace{-4pt}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%  Summary  %%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Summary}
\small{{{SUMMARY}}}
\vspace{-4pt}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%  Experience  %%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Experience}
  \resumeSubHeadingListStart
{{EXPERIENCE}}
  \resumeSubHeadingListEnd

%%%%%%%%%%%%%%%%%%%%%%%%%%%%  PROJECTS  %%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Projects}
\resumeSubHeadingListStart
{{PROJECTS}}
\resumeSubHeadingListEnd

%%%%%%%%%%%%%%%%%%%%%%%%%%%%  AWARDS  %%%%%%%%%%%%%%%%%%%%%%%%%%%%
% build-cv-latex.mjs emits {{AWARDS}} as \projheading lines, which expand to
% \resumeProjectHeading and therefore open with \item — so unlike Education this
% one DOES need the itemize wrapper. Without this block the emitter's awards
% substitution had nowhere to land and honours were silently dropped.
\section{Awards \& Honors}
\resumeSubHeadingListStart
{{AWARDS}}
\resumeSubHeadingListEnd

%%%%%%%%%%%%%%%%%%%%%%%%%%%%  Education  %%%%%%%%%%%%%%%%%%%%%%%%%%%%
% No \resumeSubHeadingListStart wrapper here, unlike Experience and Projects.
% Those two are emitted starting with \roleheading / \projheading, which expand
% to \resumeSubheading / \resumeProjectHeading and therefore begin with \item —
% legal inside an itemize. Education is emitted as a bare tabular* with no
% \item, so wrapping it produced "Something's wrong--perhaps a missing \item"
% and no PDF at all.
\section{Education}
{{EDUCATION}}

%%%%%%%%%%%%%%%%%%%%%%%%%%%%  Technical Skills  %%%%%%%%%%%%%%%%%%%%%%%%%%%%
\section{Technical Skills}
\vspace{-7pt}
\begin{itemize}
[leftmargin=0.15in, label={}]\small{\item{
{{SKILLS}}
}}
\end{itemize}
